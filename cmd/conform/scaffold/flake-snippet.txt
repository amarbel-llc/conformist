Add conformist to flake.nix and wire it into your per-system outputs
(utils.lib.eachDefaultSystem or a flake-parts perSystem):

  inputs = {
    conformist.url = "git+https://code.linenisgreat.com/conformist.git";
    # Dedupe conformist's inputs against your same-named ones from INSIDE the
    # conformist input — never add a top-level input your outputs pattern does
    # not name, or a strict destructuring fails eval (conformist#83):
    conformist.inputs.igloo.follows = "igloo";
    conformist.inputs.nixpkgs-master.follows = "nixpkgs-master";
    conformist.inputs.utils.follows = "utils";
    # just-us is the eng `just` fork: it supplies BOTH the devShell's `just`
    # and the justfile-orphan-summary linter module, which reads a fork-only
    # field upstream `just` never emits. Drop a follows line for any input you
    # do not have — a follows pointing at a missing input fails eval too:
    just-us.url = "git+https://code.linenisgreat.com/just-us.git";
    just-us.inputs.nixpkgs.follows = "nixpkgs";
    just-us.inputs.flake-utils.follows = "utils";
    just-us.inputs.conformist.follows = "conformist";
  };

  # per system, with `pkgs` and the `conformist` / `just-us` inputs in scope:
  let
    conformistPkg = conformist.packages.${system}.default;
    justPkg = just-us.packages.${system}.default;
    eval = conformist.lib.evalModule pkgs {
      imports = [
        conformist.lib.presets.eng   # the eng-convention roster (no justfile rules)
        # the justfile-* convention rules, shipped from just-us because they
        # read a fork-only `just` dump format; enables all eight:
        just-us.lib.conformistPresets.justfile
        ./conformist.nix             # this repo's overlay (written by `conform`)
      ];
      package = conformistPkg;

      # one setting for the whole justfile-linter family; MUST be the fork
      linters.justfile-common.justPackage = justPkg;
    };
  in {
    formatter = eval.config.build.wrapper;             # `nix fmt`
    checks.formatting = eval.config.build.check self;  # read-only `conformist check` gate

    # impure git-state lane (git-remotes, sweatfile, agents-md) for `lint-worktree`:
    packages.conformist-impure-config =
      (conformist.lib.evalModule pkgs {
        imports = [ conformist.lib.presets.eng-impure ];
        package = conformistPkg;
        projectRootFile = "flake.nix";
      }).config.build.configFile;

    # git pre-commit + (opt-in) merge-time repair hooks for THIS repo's config,
    # on the devShell PATH as `conformist-pre-commit` / `conformist-repair` (the
    # sweatfile names them). See conformist#47/#54 and conformist.7 PRE-COMMIT.
    packages.conformist-pre-commit = eval.config.build.preCommit;
    packages.conformist-repair = eval.config.build.repair;
    devShells.default = pkgs.mkShell {
      packages = [
        conformistPkg
        eval.config.build.preCommit
        eval.config.build.repair
        justPkg
      ];
    };
  }
